static void
testlv13c_probe_5d4(struct event *event)
{
    if (!event)
        return;

    if (!menu_shown)
        return;

    volatile struct testlv13c_fp_5d4 *st = 0;

    if (testlv6_lv_menu_no_host_5d4)
    {
        st = &testlv13c_lv_5d4;
    }
    else if (
        !lv_5d4 &&
        CURRENT_GUI_MODE == GUIMODE_PLAY
    )
    {
        st = &testlv13c_m1_5d4;
    }
    else
    {
        return;
    }

    uint32_t param =
        (uint32_t) event->param;

    uint32_t arg =
        (uint32_t) event->arg;

    /*
     * MAIN dial candidates:
     *
     * M1 may already receive native logical LEFT/RIGHT.
     *
     * LV is known to receive:
     *     param 0x61
     *     arg   0x0B
     */
    if (
        param != BGMT_WHEEL_LEFT &&
        param != BGMT_WHEEL_RIGHT &&
        !(
            param == 0x61 &&
            arg == 0x0B
        )
    )
    {
        return;
    }

    st->n++;

    st->last_param = param;
    st->last_arg = arg;
    st->last_obj =
        (uint32_t) event->obj;

    if (param == BGMT_WHEEL_LEFT)
    {
        st->logical_left++;
    }
    else if (param == BGMT_WHEEL_RIGHT)
    {
        st->logical_right++;
    }
    else
    {
        st->raw61_main++;

        /*
         * Same payload-safety rule already HW-used by TESTLV7C.
         *
         * READ ONLY.
         */
        uint32_t objp =
            (uint32_t) event->obj;

        if (
            event->obj != NULL &&
            !(objp & 0xF0000000) &&
            objp >= 0x4000
        )
        {
            volatile uint32_t *q =
                (volatile uint32_t *) event->obj;

            st->w0 = q[0];
            st->w1 = q[1];
            st->w2 = q[2];

            st->payload_valid = 1;
        }
        else
        {
            st->payload_valid = 0;
        }
    }
}

static void level_ml_entry_failsafe_cbr()
{
    if (!level_ml_entry_armed_5d4)
        return;

    /*
     * Normal N-B completes far before this callback.
     * If anything went wrong, stop the transaction and reveal
     * Canon rather than leaving the LCD permanently black.
     */
    level_ml_entry_stage_5d4 = 0;
    level_ml_entry_poll_count_5d4 = 0;
    ml_menu_level_host_prepared_5d4 = 0;

    if (level_curtain_active_5d4)
        level_curtain_disable_5d4();

    level_ml_entry_armed_5d4 = 0;
}

/*
 * Establish Canon's backing GUI while ML is still closed.
 *
 * Stage 1 waits for synthetic Q to commit mode 0x29.
 * Stage 2 requests GUIMODE_PLAY and waits for that mode to commit.
 * Only then is gui_sem released to open the ML menu.
 */
static void level_ml_entry_cbr()
{
    if (!level_ml_entry_stage_5d4)
        return;

    /*
     * Absolutely no Canon host transition after ML opens.
     */
    if (menu_shown || lv_5d4)
    {
        level_ml_entry_stage_5d4 = 0;
        level_ml_entry_poll_count_5d4 = 0;
        ml_menu_level_host_prepared_5d4 = 0;
        return;
    }

    /*
     * Stage 1:
     * wait until fake Q has really committed Quick Control.
     */
    if (level_ml_entry_stage_5d4 == 1)
    {
        if (CURRENT_GUI_MODE == 0x29)
        {
            /*
             * TESTLV13B B:
             * Quick Control committed.
             */
            testlv13b_capture_5d4(
                &testlv13b_b_5d4
            );

            /*
             * TESTLV14B phase C:
             * CURRENT_GUI_MODE == 0x29.
             *
             * PLAY will now be requested while this phase remains
             * active, until Canon commits GUIMODE_PLAY.
             */
            testlv14b_phase_5d4 = 3;

            level_ml_entry_stage_5d4 = 2;
            level_ml_entry_poll_count_5d4 = 0;

            
            SetGUIRequestMode(GUIMODE_PLAY);

            /*
             * TESTLV13B C:
             * immediately after PLAY request.
             */
            testlv13b_capture_5d4(
                &testlv13b_c_5d4
            );

            delayed_call(
                10,
                level_ml_entry_cbr,
                0
            );

            return;
        }

        level_ml_entry_poll_count_5d4++;

        if (level_ml_entry_poll_count_5d4 < 50)
        {
            delayed_call(
                10,
                level_ml_entry_cbr,
                0
            );
            return;
        }

        level_ml_entry_stage_5d4 = 0;
        level_ml_entry_poll_count_5d4 = 0;
        return;
    }

    /*
     * Stage 2:
     * PLAY has already been requested, still with ML closed.
     *
     * AUDIT100 showed CURRENT commit alone is not a proof that
     * every late PLAY redraw has completed, therefore CURRENT==2
     * is followed by a short explicit settle stage.
     */
    if (level_ml_entry_stage_5d4 == 2)
    {
        if (CURRENT_GUI_MODE == GUIMODE_PLAY)
        {
            /*
             * TESTLV13B D:
             * PLAY committed.
             */
            testlv13b_capture_5d4(
                &testlv13b_d_5d4
            );

            /*
             * TESTLV14B phase D:
             * PLAY committed, ML still closed.
             */
            testlv14b_phase_5d4 = 4;

            
            level_ml_entry_stage_5d4 = 0;
            level_ml_entry_poll_count_5d4 = 0;

            ml_menu_level_host_prepared_5d4 = 1;

            give_semaphore(gui_sem);
            return;
        }

        level_ml_entry_poll_count_5d4++;

        if (level_ml_entry_poll_count_5d4 < 50)
        {
            delayed_call(
                10,
                level_ml_entry_cbr,
                0
            );
            return;
        }

        level_ml_entry_stage_5d4 = 0;
        level_ml_entry_poll_count_5d4 = 0;
        return;
    }

    level_ml_entry_stage_5d4 = 0;
    level_ml_entry_poll_count_5d4 = 0;
}
